/*
 * P9 harness: runs the driver's hwmon read path (temp*_input, fan*_input)
 * against a fake ACPI EC (0x62/0x66, read command 0x80) and a fake PMC2
 * (0x68/0x6c, 0xd5 + tach subcommands 0x14-0x19) in userspace and checks
 * the 0009 cache: hits without port I/O, refresh after 1 s, errors not
 * cached, per-input refresh, dead flag, killable lock, pwm path unchanged.
 * Build:
 *   sed '/^#include <linux\//d' minisforum_n5_it5571.c > drv.c
 *   gcc -O0 -g -Wall -Wno-unused-function -DDRV='"drv.c"' h9.c -o h9
 * Fake EC model: the EC answers every read at once; "stuck" keeps IBF set
 * on both interfaces (every wait times out within the access budget).
 */
#include "stubs9.h"

s64 fake_ns;
int h_lock_violation;
int h_kill;
struct platform_device h_pdev;

#include DRV

/* ---------------- fake EC ---------------- */
static u8 ec_ram[256];
static u16 tach[3];
static int ec_state, ec_obf;
static u8 ec_out;
static int pmc_state, pmc_obf, cur_sub;
static u8 pmc_out;
static int ec_reads, pmc_reads, pmc_writes;
static int io_count, io_lock_bad, io_region_bad;
static int stuck;
static unsigned regions_held;
static struct n5_data *g;

void hlog(const char *fmt, ...) { (void)fmt; }

static unsigned port_bit(unsigned short p)
{
	switch (p) {
	case 0x62: return 1; case 0x66: return 2; case 0x68: return 4; case 0x6c: return 8;
	}
	return 0;
}

static void io_check(unsigned short port)
{
	unsigned b = port_bit(port);

	io_count++;
	if (!g || !g->lock.locked)
		io_lock_bad++;
	if (b && !(regions_held & b))
		io_region_bad++;
}

static int is_write_sub(int s)
{
	static const int w[] = { 0x23, 0x25, 0x27, 0x29, 0x2f, 0x2c, 0x33, 0x35, 0x37, 0x39 };
	unsigned i;

	for (i = 0; i < ARRAY_SIZE(w); i++)
		if (w[i] == s)
			return 1;
	return 0;
}

static u8 tach_byte(int sub)
{
	/* low 0x18/0x16/0x14, high 0x19/0x17/0x15 for fan1/fan2/fan3 */
	int ch = (0x19 - sub) / 2;
	int high = sub & 1;

	return high ? tach[ch] >> 8 : tach[ch] & 0xff;
}

void h_outb(u8 v, unsigned short port)
{
	io_check(port);
	if (stuck)
		return;
	if (port == 0x66 && v == 0x80) {
		ec_state = 1;
		return;
	}
	if (port == 0x62 && ec_state == 1) {
		ec_out = ec_ram[v];
		ec_obf = 1;
		ec_state = 0;
		ec_reads++;
		return;
	}
	if (port == 0x6c && v == 0xd5) {
		pmc_state = 1;
		return;
	}
	if (port != 0x68)
		return;
	if (pmc_state == 1) {
		cur_sub = v;
		if (v >= 0x14 && v <= 0x19) {
			pmc_out = tach_byte(v);
			pmc_obf = 1;
			pmc_reads++;
			pmc_state = 0;
		} else if (is_write_sub(v)) {
			pmc_state = 2;
		} else {
			pmc_state = 0; /* action */
		}
	} else if (pmc_state == 2) {
		pmc_writes++;
		pmc_state = 0;
	}
}

u8 h_inb(unsigned short port)
{
	io_check(port);
	switch (port) {
	case 0x66:
		return (stuck ? 0x02 : 0) | (ec_obf ? 0x01 : 0);
	case 0x62:
		ec_obf = 0;
		return ec_out;
	case 0x6c:
		return (stuck ? 0x02 : 0) | (pmc_obf ? 0x01 : 0);
	case 0x68:
		pmc_obf = 0;
		return pmc_out;
	}
	return 0;
}

/* ---------------- devres / DMI / hwmon ---------------- */
enum { R_REGION, R_ACTION, R_HWMON };
static struct { int kind; void (*fn)(void *); void *arg; unsigned bit; } dr[32];
static int ndr;
static struct device hwdev;

void *devm_kzalloc(struct device *d, size_t n, int gfp)
{
	(void)d; (void)gfp;
	return calloc(1, n);
}

void *devm_request_region(struct device *d, unsigned long start, unsigned long n, const char *name)
{
	(void)d; (void)n; (void)name;
	dr[ndr].kind = R_REGION;
	dr[ndr].bit = port_bit(start);
	regions_held |= dr[ndr].bit;
	ndr++;
	return (void *)1;
}

int devm_add_action_or_reset(struct device *d, void (*fn)(void *), void *arg)
{
	(void)d;
	dr[ndr].kind = R_ACTION;
	dr[ndr].fn = fn;
	dr[ndr].arg = arg;
	ndr++;
	return 0;
}

struct device *devm_hwmon_device_register_with_info(struct device *d, const char *name,
	void *drvdata, const struct hwmon_chip_info *ci, const void *groups)
{
	(void)d; (void)name; (void)ci; (void)groups;
	hwdev.drvdata = drvdata;
	dr[ndr++].kind = R_HWMON;
	return &hwdev;
}

const struct dmi_system_id *h_dmi_first_match(const struct dmi_system_id *t)
{
	return &t[1]; /* N5 PRO / F8NAA entry */
}

int h_dmi_match(int f, const char *s)

	(void)f; (void)s;

/* ---------------- test helpers ---------------- */
static int fails, passes;

#define CHECK(name, cond) do { \
	if (cond) { passes++; printf("PASS  %s\n", name); } \
	else { fails++; printf("FAIL  %s  (line %d)\n", name, __LINE__); } \
} while (0)

static void probe(void)
{
	ndr = 0;
	regions_held = 0;
	h_pdev.drvdata = NULL;
	experimental_write = 1;
	if (n5_probe(&h_pdev))
		printf("probe failed\n");
	g = h_pdev.drvdata;
}

static void unbind(void)
{
	int i;

	for (i = ndr - 1; i >= 0; i--) {
		if (dr[i].kind == R_REGION)
			regions_held &= ~dr[i].bit;
		else if (dr[i].kind == R_ACTION)
			dr[i].fn(dr[i].arg);
	}
	ndr = 0;
}

static int rt(int ch, long *v) { return n5_read(&hwdev, hwmon_temp, hwmon_temp_input, ch, v); }
static int rf(int ch, long *v) { return n5_read(&hwdev, hwmon_fan, hwmon_fan_input, ch, v); }
static void ms(int n) { fake_ns += (s64)n * 1000000LL; }

int main(void)
{
	long v, v2;
	int ret, n, i, ok;
	s64 t0;

	fake_ns = 5000000000LL; /* 5 s after boot */
	memset(ec_ram, 0, sizeof(ec_ram));
	ec_ram[0x09] = 45; ec_ram[0x04] = 33; ec_ram[0x05] = 30; ec_ram[0x06] = 22;
	tach[0] = 2000; tach[1] = 2200; tach[2] = 1240;

	printf("== C1 temp1: first read from EC, second from cache, refresh after 1 s\n");
	probe();
	CHECK("C1 probe does no port I/O", io_count == 0);
	n = ec_reads;
	CHECK("C1 temp1 = 45000", rt(0, &v) == 0 && v == 45000);
	CHECK("C1 one EC read", ec_reads == n + 1);
	ec_ram[0x09] = 50;
	i = ec_reads;
	n = io_count;
	ms(999);
	CHECK("C1 999 ms later: cached 45000", rt(0, &v) == 0 && v == 45000);
	CHECK("C1 cache hit: no port I/O", io_count == n);
	ms(2);
	CHECK("C1 1001 ms later: refreshed 50000", rt(0, &v) == 0 && v == 50000);
	CHECK("C1 refresh: one EC read", ec_reads == i + 1);

	printf("== C2 per input: temp2 not served from temp1, fan1 own cache\n");
	n = ec_reads;
	CHECK("C2 temp2 = 33000", rt(1, &v) == 0 && v == 33000);
	CHECK("C2 temp2 did its own EC read", ec_reads == n + 1);
	n = pmc_reads;
	CHECK("C2 fan1 = 2000", rf(0, &v) == 0 && v == 2000);
	CHECK("C2 fan1: 3 PMC2 reads (high, low, high)", pmc_reads == n + 3);
	tach[0] = 2100;
	n = io_count;
	CHECK("C2 fan1 cached 2000", rf(0, &v) == 0 && v == 2000);
	CHECK("C2 fan1 cache hit: no port I/O", io_count == n);
	n = pmc_reads;
	CHECK("C2 fan2 = 2200 (own refresh)", rf(1, &v) == 0 && v == 2200 && pmc_reads == n + 3);
	ms(1001);
	CHECK("C2 fan1 refreshed 2100", rf(0, &v) == 0 && v == 2100);

	printf("== C3 flood: 10000 reads of fan1 and temp1 within 1 s\n");
	ms(1001);
	n = pmc_reads;
	i = ec_reads;
	ok = 1;
	for (ret = 0; ret < 10000; ret++) {
		if (rf(0, &v) || v != 2100 || rt(0, &v2) || v2 != 50000)
			ok = 0;
		fake_ns += 50000; /* 50 us per pair: 0.5 s in total */
	}
	CHECK("C3 all values correct", ok);
	CHECK("C3 fan1: one refresh (3 PMC2 reads)", pmc_reads == n + 3);
	CHECK("C3 temp1: one refresh (1 EC read)", ec_reads == i + 1);

	printf("== C4 errors are not cached\n");
	ms(1001);
	stuck = 1;
	t0 = fake_ns;
	ret = rt(0, &v);
	CHECK("C4 stuck EC: temp1 -ETIMEDOUT", ret == -ETIMEDOUT);
	CHECK("C4 within access budget (<= 1.02 s)", fake_ns - t0 <= 1020000000LL);
	CHECK("C4 stale value not served after failed refresh", rt(0, &v) == -ETIMEDOUT);
	stuck = 0;
	ec_ram[0x09] = 47;
	n = ec_reads;
	CHECK("C4 EC back: next read goes to the EC (47000)", rt(0, &v) == 0 && v == 47000 && ec_reads == n + 1);
	stuck = 1;
	ms(1001);
	CHECK("C4 fan1 stuck -> -ETIMEDOUT", rf(0, &v) == -ETIMEDOUT);
	stuck = 0;
	n = pmc_reads;
	CHECK("C4 fan1 retried at once", rf(0, &v) == 0 && v == 2100 && pmc_reads > n);

	printf("== C6 sentinels: cached raw, still -ENODATA\n");
	ms(1001);
	ec_ram[0x05] = 0xff;
	n = ec_reads;
	CHECK("C6 temp3 0xff -> -ENODATA", rt(2, &v) == -ENODATA && ec_reads == n + 1);
	CHECK("C6 temp3 again -> -ENODATA from cache", rt(2, &v) == -ENODATA && ec_reads == n + 1);
	tach[2] = 0xffff;
	n = pmc_reads;
	CHECK("C6 fan3 0xffff -> -ENODATA", rf(2, &v) == -ENODATA && pmc_reads == n + 3);
	CHECK("C6 fan3 again -> -ENODATA from cache", rf(2, &v) == -ENODATA && pmc_reads == n + 3);
	tach[2] = 0;
	ms(1001);
	CHECK("C6 fan3 0 rpm is a value (0)", rf(2, &v) == 0 && v == 0);
	ec_ram[0x05] = 30;

	printf("== C7 killable: a killed reader gets -EINTR, no I/O, no state\n");
	ms(1001);
	h_kill = 1;
	n = io_count;
	CHECK("C7 temp1 -EINTR", rt(0, &v) == -EINTR);
	CHECK("C7 fan1 -EINTR", rf(0, &v) == -EINTR);
	CHECK("C7 no port I/O", io_count == n);
	CHECK("C7 lock free afterwards", !g->lock.locked);
	h_kill = 0;

	printf("== C8 pwm path unchanged: write, cached pwm read, fan cache not invalidated\n");
	CHECK("C8 fan1 refresh", rf(0, &v) == 0 && v == 2100);
	n = pmc_writes;
	CHECK("C8 pwm1_enable=1 ok", n5_write(&hwdev, hwmon_pwm, hwmon_pwm_enable, 0, 1) == 0);
	CHECK("C8 pwm1=128 ok", n5_write(&hwdev, hwmon_pwm, hwmon_pwm_input, 0, 128) == 0);
	CHECK("C8 pwm writes reached the EC", pmc_writes > n);
	CHECK("C8 pwm1 reads 128", n5_read(&hwdev, hwmon_pwm, hwmon_pwm_input, 0, &v) == 0 && v == 128);
	tach[0] = 1500;
	CHECK("C8 fan1 still cached (2100) within 1 s after the write", rf(0, &v) == 0 && v == 2100);
	ms(1001);
	CHECK("C8 fan1 1500 after 1 s", rf(0, &v) == 0 && v == 1500);

	printf("== C9 dead: after .shutdown cached reads give -ENODEV, no I/O\n");
	CHECK("C9 temp1 cached before shutdown", rt(0, &v) == 0);
	n5_shutdown(&h_pdev);
	n = io_count;
	CHECK("C9 temp1 -ENODEV", rt(0, &v) == -ENODEV);
	CHECK("C9 fan1 -ENODEV", rf(0, &v) == -ENODEV);
	CHECK("C9 no port I/O after shutdown", io_count == n);
	unbind();

	printf("== C10 rebind: cache starts empty\n");
	probe();
	n = ec_reads;
	CHECK("C10 temp1 read goes to the EC", rt(0, &v) == 0 && ec_reads == n + 1);
	n = io_count;
	unbind();
	CHECK("C10 unbind: no I/O (nothing touched)", io_count == n);
	n = io_count;
	CHECK("C10 read after unbind -> -ENODEV", rt(0, &v) == -ENODEV && io_count == n);

	printf("== global\n");
	CHECK("every port I/O under data->lock", io_lock_bad == 0);
	CHECK("every port I/O inside reserved regions", io_region_bad == 0);
	CHECK("no mutex misuse", h_lock_violation == 0);
	printf("RESULT %d pass, %d fail\n", passes, fails);
	return fails ? 1 : 0;
}
