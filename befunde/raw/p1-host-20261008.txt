### Skripttext
#!/bin/bash
# P1 host measurement - READ ONLY
run() { echo; echo "\$ $*"; bash -c "$*" 2>&1; echo "[rc=$?]"; }
H=$(hostname)
echo "### 1. Kernel"
echo; echo '$ uname -a'; uname -a | sed "s/$H/n5host/g"; echo "[rc=$?]"
run 'uname -r'
run 'grep -E "^(# )?CONFIG_(DEBUG_FS|DEBUG_FS_ALLOW_ALL|DEBUG_FS_DISALLOW_MOUNT|DEBUG_FS_ALLOW_NONE|SECURITY_LOCKDOWN_LSM|SECURITY_LOCKDOWN_LSM_EARLY|LOCK_DOWN_KERNEL_FORCE_NONE|LOCK_DOWN_KERNEL_FORCE_INTEGRITY|LOCK_DOWN_KERNEL_FORCE_CONFIDENTIALITY|MODULE_SIG|MODULE_SIG_FORCE|MODULE_SIG_ALL|LSM|IO_STRICT_DEVMEM|STRICT_DEVMEM|DEVPORT)[= ]" /boot/config-$(uname -r)'
echo; echo "### 2. debugfs / lockdown / cmdline"
run 'findmnt /sys/kernel/debug'
run 'findmnt -t debugfs'
run 'sudo -n cat /sys/kernel/security/lockdown'
run 'cat /proc/cmdline'
echo; echo "### 3. ioports / lsmod"
run 'sudo -n cat /proc/ioports | grep -iE "002e-002f|004e-004f|0062|0066|0068|006c|it87|nct|w83627|f71882|sch5|minisforum|ec"'
run 'lsmod | grep -iE "it87|nct6775|nct6683|w83627|f71882|sch5|ec_sys|acpi_ec|minisforum"'
echo; echo "### 4. Modul"
run 'modinfo minisforum_n5_it5571'
for f in version srcversion taint refcnt initstate; do run "cat /sys/module/minisforum_n5_it5571/$f"; done
run 'ls -l /sys/module/minisforum_n5_it5571/parameters/'
run 'for p in /sys/module/minisforum_n5_it5571/parameters/*; do echo "$p = $(sudo -n cat $p)"; done'
run 'sudo -n dkms status'
run 'command -v mokutil && mokutil --sb-state'
run 'sudo -n journalctl -k -b --no-pager | grep -i minisforum'
run 'cat /proc/sys/kernel/tainted'
echo; echo "### 5. n5-fangov"
run 'systemctl cat n5-fangov'
run 'systemctl show n5-fangov -p ActiveState,SubState,ActiveEnterTimestamp,NRestarts,WatchdogUSec,KillMode,Restart,ExecStopPost,ExecStartPre,OnFailure,MainPID'
run 'dpkg -l | grep -E "n5-fangov|minisforum"'
echo; echo "### 6. hwmon"
run 'for d in /sys/class/hwmon/hwmon*; do echo "$d $(cat $d/name)"; done'
D=$(for d in /sys/class/hwmon/hwmon*; do n=$(cat $d/name); case "$n" in *minisforum*|*n5*|*it5571*) echo $d;; esac; done | head -1)
echo; echo "# minisforum hwmon dir: $D"
run "ls -l $D/"
run "cd $D && for f in name pwm1 pwm2 pwm3 pwm4 pwm1_enable pwm2_enable pwm3_enable pwm4_enable fan1_input fan2_input fan3_input fan4_input temp1_input temp2_input temp3_input temp4_input temp1_label temp2_label temp3_label temp4_label; do if [ -e \$f ]; then echo \"\$f = \$(cat \$f 2>&1)\"; else echo \"\$f = (nicht vorhanden)\"; fi; done"
run "readlink -f $D/device"
